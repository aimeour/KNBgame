def checkentr(event): #проверка данных для входа

    login = logentr.get() #переменная, содержащая логин
    #пароль
    password = passwordentr.get() #переменная, содержащая пароль
    #проверка по логину
    #выбираем столбец логина, а оттуда те логины, которые соответствуют введенному 
    sql.execute(f"SELECT login, password FROM users WHERE login = '{login}' AND password = '{password}'") 
    #если таких нет, то check = 0 
    if sql.fetchone() is None:
        check = 0
    if check == 0: #если логин не подошел, проверяем почту
        #выбираем столбец почту, а оттуда те почты, которые соответствуют введенной

def checkreg(event): #проверка данных для регистрации
    global logreg, passwordreg, mailreg, agereg, genderreg #данные
    global warnllen, warnlform, warnplen, warnpform, warnmail, warnacexstreg #вывод ошибок
    #убираем предыдущие ошибки
    warnllen.place_forget()
    warnlform.place_forget()
    warnplen.place_forget()
    warnpform.place_forget()
    warnmail.place_forget()
    warnacexstreg.place_forget()
    #проверки данных на соотвестствия некоторым шаблонам
    checkin = 1 #проверка верности ввода данных
    #логин
    login = logreg.get() #переменная, содержащая логин
    #проверка соответствия введенного логина нужному формату
    if (login.isalnum() == 1 and login.isdigit() == 0) == 0: 
        checkin = 0
        warnlform.place(relx = 0.88, rely = 0.32, anchor = CENTER)
    if len(login)<3 or len(login)>12: #проверка длинны логина
        warnlform.place_forget()
        checkin = 0
        warnllen.place(relx = 0.87, rely = 0.32, anchor = CENTER)
    #пароль
    password = passwordreg.get() #переменная, содержащая пароль
    #проверка соответствия введенного пароля нужному формату
    if password.isalnum() == 0 or password.isdigit() == 1 or password.isalpha() == 1:
        checkin = 0
        warnpform.place(relx = 0.88, rely = 0.4, anchor = CENTER)
    if len(password)<6 or len(password)>20: #проверка длинны пароля
        warnpform.place_forget()
        checkin = 0
        warnplen.place(relx = 0.88, rely = 0.4, anchor = CENTER)
    #почта
    alf = "abcdefghigklmnopqrstuvwxyzABCDEFGHIGKLMNOPQRSTUVWXYZ@.0123456789" #допустимые для почты символы
    mail = mailreg.get()
    mailcheck = 1
    for i in range(len(mail)): #проверка того все ли символы допустимы
        if mail[i] in alf == 0: #если символ не входит в алфавит, выводит ощибку ввода и останавливается
            checkin = 0
            mailcheck = 0 #чтобы не проверять подходит ли по шаблону после этого
            warnmail.place(relx = 0.87, rely = 0.48, anchor = CENTER) #вывод ошибки ввода
            break 
    if mailcheck == 1: #если символы подходят
        try:
            pat = re.compile(r"(\w+@\w+\.\w+)") #сбока регулярного выражения 
            if mail != re.fullmatch(pat, mail).group(): #если не совпадает найденная часть и исходные данные
                checkin = 0
                warnmail.place(relx = 0.87, rely = 0.48, anchor = CENTER) #вывод ошибки ввода
            else:
                try:
                    list = mail.split("@") #список с частями почты (перед @ и после)
                    list2 = list[1].split(".") #список с разделенной второй частью
                    list[1] = list2[0] #объединение списков
                    list.append(list2[1])
                    #если какая-то из частей слишком большая или маленькая
                    if (len(list[0])<3 or len(list[0])>20) or (len(list[1])<3 or len(list[1])>10) or (len(list[2])<2 or len(list[2])>4):
                        checkin = 0
                        warnmail.place(relx = 0.87, rely = 0.48, anchor = CENTER) #вывод ошибки ввода
                except IndexError:
                    checkin = 0
                    warnmail.place(relx = 0.87, rely = 0.48, anchor = CENTER) #вывод ошибки ввода
        except AttributeError: #если не подходит (тип None)
            checkin = 0
            warnmail.place(relx = 0.87, rely = 0.48, anchor = CENTER) #вывод ошибки ввода
    #возраст 
    age = agereg.get()
    #пол
    if genderreg.get() == 1:
        gender = 'жен'
    if genderreg.get() == 2:
        gender = 'муж'
    #проверка что такие данные еще не внесены в бд
    if checkin == 1:
        #выбираем столбец логина, а оттуда те логины, которые соответствуют введенному 
        sql.execute(f"SELECT login FROM users WHERE login = '{login}'") 
        #если таких нет, то добавляем данные 
        if sql.fetchone() is None:
            #выбираем столбец логина, а оттуда те логины, которые соответствуют введенному 
            sql.execute(f"SELECT mail FROM users WHERE mail = '{mail}'") 
            #если таких нет, то добавляем данные 
            if sql.fetchone() is None:
                sql.execute("INSERT INTO users VALUES (?, ?, ?, ?, ?, ?, ?, ?)", (login, password, mail, age, gender, 0, 0, 0))
                bd.commit() #подтверждение действия
            else:
                warnacexstreg.place(relx = 0.5, rely = 0.26, anchor = CENTER)
                checkin = 0
        else:
            warnacexstreg.place(relx = 0.5, rely = 0.26, anchor = CENTER)
            checkin = 0
    if checkin == 1:
        entrancef(event)

def savef(event): #сохранение данных
    global login, gamecount, wincount
    nowd = date.today() #дата 
    now = datetime.now() #дата и время
    nowt = now.strftime("%H:%M:%S") #красивый формат времени
    name = now.strftime("%H-%M %d-%m-%Y") #полная дата в простом формате для имени файла
    name += '.txt' #добавление раширения
    file_path = filedialog.asksaveasfilename(filetypes=(("Text files", "*.txt"), ("All files", "*.*")), defaultextension='.txt', initialfile = name) #способ и тип сохранения результата
    if file_path: 
        with open(file_path, "w") as file: #запись в файл
            file.write(f"Статистика игры {login} за {nowd} {nowt}\n") 
            file.write(f"Текущая игра\n")
            file.write(f"Количество игр: {gamecount}\n") 
            file.write(f"Количество побед: {wincount}\n") 
            file.write(f"Общая статистика\n")
            for value in sql.execute(f"SELECT games FROM users WHERE login = '{login}'"):
                games = int(value[0]) #получение количества игр пользователя
            file.write(f"Общее количество игр: {games}\n")
            for value in sql.execute(f"SELECT wins FROM users WHERE login = '{login}'"):
                wins = int(value[0]) #получение количества побед пользователя 
            file.write(f"Общее количество побед: {wins}\n") 
            for value in sql.execute(f"SELECT score FROM users WHERE login = '{login}'"):
                score = int(value[0]) #получение счета пользователя
            file.write(f"Счет: {score}\n")
            i = 1
            for log in sql.execute("SELECT login FROM users ORDER BY score DESC"): #определение номера в рейтинге
                if "".join(log) == login:
                    break   
                i += 1 #шаг
            file.write(f"Место в рейтинге: {i}\n")
